   first match while earlier courts are still playing. The visible CURRENT round

function currentActiveRound(rounds){

  const firstOpen=rounds.findIndex(matches=>matches.some(m=>m.status==='active'));
  return firstOpen>=0?firstOpen+1:rounds.length;
}

function lineupConflict(slot,playerId){
 if(!playerId)return null;
 const waves=lineupWaveMap(),currentWave=waves.get(slot.dataset.lineup)||1;
 return Array.from(app.querySelectorAll('select[data-lineup]')).find(other=>
  other!==slot && other.dataset.club===slot.dataset.club &&
  (waves.get(other.dataset.lineup)||1)===currentWave &&
  other.value===playerId
 )||null;
}
function refreshStage2LineupOptions(){
 const slots=Array.from(app.querySelectorAll('select[data-lineup]'));
 if(!slots.length)return;
 for(const slot of slots){
  for(const opt of slot.options){
   if(!opt.value)continue;
   if(!opt.dataset.baseLabel)opt.dataset.baseLabel=opt.textContent.split(' · IN DUEL ')[0].split(' · USED IN DUEL ')[0];
   const occupied=lineupConflict(slot,opt.value);
   // Keep selected values visible while an older backup is being corrected;
   // new duplicate selections are blocked in the change handler below.
   opt.disabled=!!occupied&&opt.value!==slot.value;
   opt.textContent=opt.dataset.baseLabel+(occupied?` · USED IN DUEL ${occupied.dataset.lineup.split('-').pop()}`:'');
  }
 }
}
function onLineupSelectionChanged(slot){
 const occupied=lineupConflict(slot,slot.value);
 if(occupied){
  const bad=slot.value;
  const prev=lineupPriorValue.get(slot);
  const match=(state.stage==='stage3'?E.stage3Matches(state):state.stage2?.matches||[]).find(m=>m.id===slot.dataset.lineup);
  const saved=match?.[`players${slot.dataset.club}`]?.[Number(slot.dataset.player)]||'';
  slot.value=prev!==undefined&&prev!==bad?prev:saved!==bad?saved:'';
  const player=state.roster[slot.dataset.club].flatMap(pair=>pair.players).find(p=>p.id===bad);
  notify(`${player?.name||'Player'} is already selected in Duel ${occupied.dataset.lineup.split('-').pop()} (same wave).`,true);
 }
 lineupPriorValue.set(slot,slot.value);
 refreshStage2LineupOptions();
}

function saveAllLineups(kind){
 if((kind==='duel'&&state.stage!=='stage2')||(kind==='final'&&state.stage!=='stage3'))throw Error('Open the correct Team Duel stage first');
 const current=kind==='duel'?state.stage2?.matches:E.stage3Matches(state);
 if(!current?.length)throw Error('No Team Duels to save');
 const scratch=JSON.parse(JSON.stringify(state));
 const proposed=current.map((match,i)=>({
  id:match.id,number:i+1,wave:match.wave||1,
  A:getLineup(kind,match.id,'A'),B:getLineup(kind,match.id,'B')
 }));
 const rosterIds={A:new Set(state.roster.A.flatMap(p=>p.players.map(x=>x.id))),B:new Set(state.roster.B.flatMap(p=>p.players.map(x=>x.id)))};
 const used={A:new Map(),B:new Map()};
 for(const p of proposed){
  for(const clubId of ['A','B']){
   const ids=p[clubId];
   if(ids.some(id=>!id||!rosterIds[clubId].has(id))||ids[0]===ids[1]){
    const msg=`Duel ${p.number}: select two different registered players for ${club(clubId)}.`;
    focusLineupError(p.id,msg);return;
   }
   for(const playerId of ids){
    const key=p.wave+':'+playerId;
    if(used[clubId].has(key)){
     const previous=used[clubId].get(key);
     const name=state.roster[clubId].flatMap(x=>x.players).find(x=>x.id===playerId)?.name||'Player';
     const msg=`Duel ${p.number}: ${name} (${club(clubId)}) already plays in Duel ${previous} of the same wave.`;
     focusLineupError(p.id,msg);return;
    }
    used[clubId].set(key,p.number);
   }
  }
 }
 // Clear provisional assignments ONLY in the scratch copy, to allow valid swaps
 // between duels without creating temporary conflicts. Production state is untouched.
 const trial=kind==='duel'?scratch.stage2.matches:E.stage3Matches(scratch);
 trial.forEach(m=>{m.playersA=['',''];m.playersB=['',''];});
 for(const p of proposed){
  try{
   if(kind==='duel')E.setDuelLineup(scratch,p.id,p.A,p.B);
   else E.setFinalLineup(scratch,p.A,p.B,p.id);
  }catch(err){focusLineupError(p.id,`Duel ${p.number}: ${err.message}`);return;}
 }
 // Restore confirmed scores ONLY when both pairs have stayed exactly the same.
 let invalidated=0,changed=0;
 for(let i=0;i<current.length;i++){
  const old=current[i],next=trial[i],p=proposed[i];
  const unchanged=['A','B'].every(c=>old[`players${c}`].every((v,j)=>v===p[c][j]));
  if(unchanged){next.scoreA=old.scoreA;next.scoreB=old.scoreB;next.winner=old.winner;}
  else {changed++;if(old.scoreA!==null||old.scoreB!==null)invalidated++;}
 }
 // E.setDuelLineup invalidates Stage 3 in scratch for any changed Stage 2 lineup.
 if(kind==='duel'&&changed===0)scratch.stage3=state.stage3;
 if(changed)scratch.finalized=false;
 const commit=()=>{state=scratch;save();render();notify(`${current.length} lineups saved together${invalidated?` · ${invalidated} changed match score(s) cleared`:''}`);};
 if(invalidated){
  confirmAction(`${invalidated} scored duel${invalidated===1?'':'s'} have changed lineups. Saving all will clear only those scores. Unchanged duel results stay confirmed. Continue?`,()=>{state=scratch;});
  return;
 }
 commit();
}
